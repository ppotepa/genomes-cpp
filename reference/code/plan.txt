13. FLOOR SOLVER — PIERWSZY ALGORYTM
======================================================================

FAZA 1:
ORTHOGONAL POLYGONS.

Pipeline:
1. reserve vertical core;
2. reserve entrance zone;
3. collect candidate split lines;
4. recursively partition;
5. assign cells to room graph;
6. merge neighbor cells gdzie trzeba;
7. adjust split positions;
8. derive shared-edge adjacency;
9. validate hard;
10. score soft;
11. repeat fixed candidate count;
12. select best.

Startowe limity:
  16 candidates
  <=64 local moves na candidate.

Derived seed:
  deriveSeed(roomGeometrySeed,'candidate/'+i)

NIGDY Math.random.

91. PSEUDOKOD FloorPlanSolver

class FloorPlanSolver {
  constructor(context){
    this.context=context;
    this.maxCandidates=context.maxCandidates||16;
    this.maxMoves=context.maxMoves||64;
  }

  solve(){
    const accepted=[];
    const failures=[];

    for(let i=0;i<this.maxCandidates;i++){
      const rng=new B.RNG(B.deriveSeed(
        this.context.seed,
        'candidate/'+i
      ));

      let candidate=this.initialCandidate(rng,i);

      for(let move=0;move<this.maxMoves;move++){
        const result=B.validateFloorCandidate(candidate,this.context);

        if(result.ok){
          candidate.score=B.scoreFloorPlan(candidate,this.context);
          break;
        }

        candidate=this.repairOrPerturb(candidate,result,rng,move);
      }

      const finalCheck=B.validateFloorCandidate(candidate,this.context);
      if(!finalCheck.ok){
        failures.push({
          candidate:i,
          failures:finalCheck.failures
        });
        continue;
      }

      candidate.score=B.scoreFloorPlan(candidate,this.context);
      accepted.push(candidate);
    }

    if(!accepted.length){
      const error=new Error('No feasible floor plan');
      error.code='NO_FEASIBLE_FLOOR_PLAN';
      error.data={failures};
      throw error;
    }

    accepted.sort((a,b)=>
      a.score-b.score ||
      a.candidateId-b.candidateId
    );

    return {
      selected:accepted[0],
      candidates:accepted.map(c=>({
        id:c.candidateId,
        score:c.score,
        scoreBreakdown:c.scoreBreakdown
      })),
      failures
    };
  }
}

Ważne:
- tie-break deterministic;
- brak performance.now w decyzjach;
- brak random retry count zależnego od machine timing.

function computeRequiredSlabVoid(stair,agent,slab){
  const required=[];

  for(const ramp of stair.navigation.ramps){
    const length=distance3(ramp.start,ramp.end);
    const count=Math.max(2,Math.ceil(length/.1));

    for(let i=0;i<=count;i++){
      const t=i/count;
      const foot=mix3(ramp.start,ramp.end,t);
      const topY=foot[1]+agent.standingHeight+agent.passageMargin;

      if(topY<=slab.bottomY)continue;

      required.push(
        circleOrSquareEnvelope(
          [foot[0],foot[2]],
          agent.radius+agent.passageMargin
        )
      );
    }
  }

  return B.Polygon.union(required);
}

function wallFromEdge(edge,storey,rooms,assembly,id){
  const dx=edge.b[0]-edge.a[0];
  const dz=edge.b[1]-edge.a[1];
  const length=Math.hypot(dx,dz);
  const tx=dx/length;
  const tz=dz/length;

  // dla CCW outer; dla interior direction trzeba ustalić owner
  const nx=tz;
  const nz=-tx;

  return {
    id,
    floor:storey.floor,
    edge:{
      a:edge.a.slice(),
      b:edge.b.slice()
    },
    length,
    baseY:storey.elevation,
    height:storey.height,
    tangent:[tx,tz],
    normal:[nx,nz],
    rooms,
    thickness:assembly.totalThickness,
    assemblyId:assembly.id,
    physicalSolidId:id,
    layers:assembly.layers,
    openings:[]
  };
}

Renderer:
  center midpoint
  yaw = atan2(tz,tx)
  geometry local along X
  rotate yaw.

function reachableSurfaces(navigation){
  const graph=new Map();

  for(const s of navigation.surfaces)
    graph.set(s.id,[]);

  for(const p of navigation.portals){
    graph.get(p.fromSurface)?.push(p.toSurface);
    graph.get(p.toSurface)?.push(p.fromSurface);
  }

  const start=navigation.outsideSurfaceId;
  const seen=new Set([start]);
  const queue=[start];

  for(let i=0;i<queue.length;i++){
    for(const next of graph.get(queue[i])||[]){
      if(seen.has(next))continue;
      seen.add(next);
      queue.push(next);
    }
  }

  return seen;
}

Każdy room.navigationSurfaceId musi być seen.
