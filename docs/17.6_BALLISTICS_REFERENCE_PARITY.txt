======================================================================
for fixture in stable order:
    nativeScalar := run(fixture, mode=Scalar, workers=1)
    nativeBatch1 := run(fixture, mode=Batch, workers=1)
    nativeBatchN := run(fixture, mode=Batch, workers=N)

    compare(nativeScalar, fixture.expected, policy)
    compare(nativeBatch1, nativeScalar, batchPolicy)
    compare(nativeBatchN, nativeBatch1, deterministicPolicy)
